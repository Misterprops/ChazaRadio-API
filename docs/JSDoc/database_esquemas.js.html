import mongoose from "mongoose";

/**
 * Esquema principal de usuario autenticado
 * Contiene credenciales y estado de verificación
 * 
 * @typedef {Object} User
 * @property {string} id - Identificador único del usuario
 * @property {string} nombre - Nombre visible del usuario
 * @property {string} contraseña - Contraseña en hash
 * @property {boolean} verificado - Indica si el usuario validó su correo
 * @property {string} correo - Email del usuario
 * @property {string} rol - Rol del usuario (ej: usuario, admin)
 */
const esquemaUser = new mongoose.Schema({
    id: String,
    nombre: String,
    contraseña: String,
    verificado: Boolean,
    correo: String,
    rol: String
});
/**Modelo del esquema esquemaUser conectado a Usuarios */
export const user = mongoose.models.user || mongoose.model("user", esquemaUser, "Usuarios");

const esquemaReg = new mongoose.Schema({

    creacion: { type: Date, default: Date.now }
});
/**Modelo del esquema esquemaReg conectado a Usuarios */
export const userReg = mongoose.models.userReg || mongoose.model("userReg", esquemaReg, "Usuarios");
