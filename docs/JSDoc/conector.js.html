/**
 * @fileoverview Servidor Express principal para ChazaRadio API
 * 
 * Punto de entrada (entry point) de la aplicación backend.
 * Configura Express con middleware, rutas y conexión a MongoDB.
 * 
 * @author ChazaRadio Team
 * @version 1.0.0
 * @module conector
 * 
 * @description
 * Configuración completa del servidor incluye:
 * - Middleware: CORS, JSON parsing, autenticación
 * - 14 endpoints REST para audio, posts, autenticación, likes
 * - Multer para manejo de file uploads
 * - Conexión a MongoDB via conectordb
 * - Servidor estático para archivos de audio en /media
 * 
 * @env
 * - PORT: Puerto del servidor (default: 3000)
 * - URI: MongoDB connection string (default: mongodb://localhost:27017/ChazaRadio)
 * - JWT_SECRET: Secret para tokens (requerido en .env)
 * - EMAIL_*: Variables para Nodemailer (requeridas en .env)
 * 
 * @example
 * // Iniciar servidor
 * npm start
 * // Servidor en http://localhost:3000
 */

import express from 'express';
import cors from 'cors';
import multer from "multer";
import swaggerUi from "swagger-ui-express";
import dotenv from "dotenv";
dotenv.config();

import { mail_verificar } from './componentes/mail_verificar.js';
import { conectardb } from './services/conectordb.js';
import { user_data, user_login, user_register } from './componentes/user_data.js';
import { delete_post, get_posts, upload_post } from './componentes/upload_post.js';
import { delete_audio, get_audios, get_sounds, upload_audio } from './componentes/audio_data.js';
import { get_poadcasts, upload_poadcast } from './componentes/poadcast_data.js';
import { get_likeList, likeControl } from './componentes/like_control.js';
import { swaggerSpec } from "./docs/swagger.js";
import { authenticateToken, reloadAccessToken } from './services/tokenServices.js';
import { request_code } from './componentes/mail_sender.js';
import { validarDeleteAudio, validarDeletePost, validarGetPosts, validarLikeControl, validarMailVerificar, validarRegistro, validarRequestCode, validarUploadAudio, validarUploadPoadcast, validarUploadPost, validarUserLogin } from './services/dataCheckServices.js';

/**
 * Aplicación Express principal
 */
const app = express();

/**
 * Puerto donde escucha el servidor
 * @type {number|string}
 * @default 3000
 */
const port = process.env.PORT || 3000;

/**
 * URI de conexión a MongoDB
 * @type {string}
 * @default "mongodb://localhost:27017/ChazaRadio"
 */
const uri = process.env.URI || "mongodb://localhost:27017/ChazaRadio";

// Middleware global
app.use(cors());
app.use(express.json());

/**
 * Inicia servidor en puerto especificado
 * @fires app#listen
 */
app.listen(port, () => {
    console.log(`Servidor Node escuchando en http://localhost:${port}`);
});

app.use("/api/docs", swaggerUi.serve, swaggerUi.setup(swaggerSpec));

app.get("/api/docs.json", (req, res) => {
    res.json(swaggerSpec);
});

/**LOCAL ONLY - Imports para manejo de archivos*/
import { fileURLToPath } from 'url';
import path from 'path';
import fs from 'fs';

/** Path de la raiz*/
const __filename = fileURLToPath(import.meta.url);
/** Nombre de la ruta de la raiz */
const __dirname = path.dirname(__filename);
/** Direccion de la carpeta */
const mediaFolder = path.resolve(__dirname, './media');

/** Asegura que la carpeta exista*/
if (!fs.existsSync(mediaFolder)) {
    fs.mkdirSync(mediaFolder, { recursive: true });
}
/** Usa la direccion de la carpeta media para audio */
app.use("/media", express.static(path.join(__dirname, "./media")));
//LOCAL ONLY

/**
 * Middleware para subir audio
 * Posee limite de 15MB
 */
const uploader = multer({
    limits: {
        fileSize: 15 * 1024 * 1024 // 15MB
    }
});

await conectardb(uri)

/**
 * @swagger
 * /api/upload:
 *   post:
 *     summary: Subir audio
 *     tags: [Audios]
 *     security:
 *       - bearerAuth: []
 *     requestBody:
 *       required: true
 *       content:
 *         multipart/form-data:
 *           schema:
 *             type: object
 *             required: [audio, data]
 *             properties:
 *               audio:
 *                 type: string
 *                 format: binary
 *               data:
 *                 type: string
 *                 example: '{"titulo":"Mi audio","tipo":"cancion","autor":"Andres"}'
 *     responses:
 *       200:
 *         description: Audio subido
 *       400:
 *         description: Datos inconsistentes
 *       500:
 *         description: Error de servidor
 */
app.post("/api/upload", authenticateToken, uploader.single("audio"), validarUploadAudio, async (req, res) => { upload_audio(req, res) }); // Revisar upload en blob

/**
 * @swagger
 * /api/get_sounds:
 *   get:
 *     summary: Obtener siguiente audio para streaming (Icecast / Liquidsoap)
 *     description: |
 *       Este endpoint devuelve una cadena de texto en formato especial utilizada por Icecast/Liquidsoap.
 *       No retorna JSON, sino texto plano con metadatos del audio actual.
 *       
 *       Formato de respuesta:
 *       annotate:title="TITULO",artist="AUTOR":URL_AUDIO
 *       
 *       Ejemplo:
 *       annotate:title="Mi canción",artist="Andres":http://localhost:3000/media/audio.mp3
 *       
 *       ⚠️ Este endpoint está diseñado para consumo por sistemas de streaming, no frontend.
 *     tags: [Streaming]
 *     responses:
 *       200:
 *         description: Cadena de streaming en texto plano
 *         content:
 *           text/plain:
 *             schema:
 *               type: string
 *               example: annotate:title="Song",artist="Artist":http://localhost:3000/media/song.mp3
 */
app.get("/api/get_sounds", async (req, res) => { get_sounds(req, res) });
